cmake_minimum_required(VERSION 3.24)
project(LivefyCameraDirectShow LANGUAGES CXX)
if(NOT DEFINED DIRECTSHOW_BASECLASSES_DIR)
  message(FATAL_ERROR "DIRECTSHOW_BASECLASSES_DIR is required")
endif()
set(BASE_SOURCES amextra.cpp amfilter.cpp amvideo.cpp arithutil.cpp combase.cpp cprop.cpp ctlutil.cpp ddmm.cpp dllentry.cpp dllsetup.cpp mtype.cpp outputq.cpp perflog.cpp pstream.cpp pullpin.cpp refclock.cpp renbase.cpp schedule.cpp seekpt.cpp source.cpp strmctl.cpp sysclock.cpp transfrm.cpp transip.cpp videoctl.cpp vtrans.cpp winctrl.cpp winutil.cpp wxdebug.cpp wxlist.cpp wxutil.cpp)
list(TRANSFORM BASE_SOURCES PREPEND "${DIRECTSHOW_BASECLASSES_DIR}/")
add_library(LivefyCameraDirectShow SHARED LivefyDirectShow.cpp Setup.cpp ${BASE_SOURCES} LivefyDirectShow.def)
target_include_directories(LivefyCameraDirectShow PRIVATE "${DIRECTSHOW_BASECLASSES_DIR}")
target_compile_features(LivefyCameraDirectShow PRIVATE cxx_std_17)
target_compile_definitions(LivefyCameraDirectShow PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(LivefyCameraDirectShow PRIVATE /W3 /EHsc /MT)
target_link_libraries(LivefyCameraDirectShow PRIVATE strmiids winmm quartz ole32 oleaut32 uuid advapi32)
set_target_properties(LivefyCameraDirectShow PROPERTIES OUTPUT_NAME "LivefyCameraDirectShow")
